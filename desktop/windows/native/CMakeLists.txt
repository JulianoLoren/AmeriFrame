cmake_minimum_required(VERSION 3.22)
project(MosaicGeometry LANGUAGES CXX)
add_library(MosaicGeometry SHARED GeometryC.cpp ../../../mobile/core/Geometry.cpp)
target_compile_features(MosaicGeometry PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(MosaicGeometry PRIVATE /fp:strict /utf-8 /W4)
    set_property(TARGET MosaicGeometry PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MosaicGeometry PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
